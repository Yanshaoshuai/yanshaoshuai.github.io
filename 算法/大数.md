# 大数相加
```java
class Solution {
    public String addStrings(String num1, String num2) {
        StringBuilder sb=new StringBuilder();
        int odd=0;
        int p1=num1.length()-1;
        int p2=num2.length()-1;
        while(p1>=0||p2>=0||odd>0){
            if(p1>=0&&p2>=0){
                int n1=num1.charAt(p1)- '0';
                int n2=num2.charAt(p2)-'0';
                int num = (n1+n2+odd)%10;
                sb.append(String.valueOf(num));
                odd = (n1+n2+odd)/10;
                p1--;
                p2--;
            }else if(p1>=0){
                int n1=num1.charAt(p1)- '0';
                int num = (n1+odd)%10;
                sb.append(String.valueOf(num));
                odd = (n1+odd)/10;
                p1--;
            }else if(p2>=0){
                int n2=num2.charAt(p2)- '0';
                int num = (n2+odd)%10;
                sb.append(String.valueOf(num));
                odd = (n2+odd)/10;
                p2--;
            }else {
                sb.append(String.valueOf(odd));
                odd=0;
            }

        }
        return sb.reverse().toString();
    }
}
```
